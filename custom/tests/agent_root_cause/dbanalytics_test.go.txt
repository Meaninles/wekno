package dbanalytics

import (
    "context"
    "database/sql"
    "errors"
    "testing"
    "time"

    _ "github.com/duckdb/duckdb-go/v2"
)

type auditConnector struct { Connector; rows []map[string]any; failAfter int; failure error }
func (a auditConnector) StreamRows(ctx context.Context, _ SourceConfig, _ TableRef, _ time.Duration, consume func(map[string]any) error) error {
    for i,row:=range a.rows {
        if err:=ctx.Err();err!=nil{return err}
        if a.failure!=nil && i==a.failAfter {return a.failure}
        if err:=consume(row);err!=nil{return err}
    }
    return nil
}
func TestAuditCompleteBatchedMaterialization(t *testing.T) {
    ctx:=context.Background(); db,err:=sql.Open("duckdb",""); if err!=nil {t.Fatal(err)}; defer db.Close()
    conn,err:=db.Conn(ctx);if err!=nil{t.Fatal(err)};defer conn.Close()
    rows:=make([]map[string]any,50001)
    for i:=range rows {score:=int64(i+1); if i==50000 {score=1000000};rows[i]=map[string]any{"id":i+1,"score":score}}
    table:=SourceTable{VirtualName:"audit_rows",Columns:[]SourceColumn{{ColumnName:"id",DataType:"integer"},{ColumnName:"score",DataType:"bigint"}}}
    start:=time.Now()
    if err:=(&Service{}).materializeTable(ctx,conn,auditConnector{rows:rows},Source{},SourceConfig{},table);err!=nil{t.Fatal(err)}
    var count,maxScore int64
    if err:=conn.QueryRowContext(ctx,"SELECT COUNT(*), MAX(score) FROM audit_rows").Scan(&count,&maxScore);err!=nil{t.Fatal(err)}
    if count!=50001||maxScore!=1000000 {t.Fatalf("incomplete aggregate: %d %d",count,maxScore)}
    t.Logf("full_source_rows=%d maximum=%d materialization_and_query_s=%.3f",count,maxScore,time.Since(start).Seconds())
    failure:=errors.New("source stream interrupted")
    if err:=(&Service{}).materializeTable(ctx,conn,auditConnector{rows:rows,failAfter:777,failure:failure},Source{},SourceConfig{},table);!errors.Is(err,failure){t.Fatalf("source failure not propagated: %v",err)}
    if err:=conn.QueryRowContext(ctx,"SELECT COUNT(*) FROM audit_rows").Scan(&count);err!=nil{t.Fatal(err)}
    if count!=0{t.Fatalf("partial rows committed: %d",count)}
    cancelled,cancel:=context.WithCancel(ctx);cancel()
    if err:=(&Service{}).materializeTable(cancelled,conn,auditConnector{rows:rows},Source{},SourceConfig{},table);err==nil{t.Fatal("cancelled import succeeded")}
}
