func auditHistoryMessages(first,last int) []*types.Message {
    var out []*types.Message
    for i:=first;i<=last;i++ {
        request:=fmt.Sprintf("audit-request-%d",i)
        out=append(out,&types.Message{ID:request+"u",RequestID:request,Role:"user",Content:fmt.Sprintf("fact_%d",i),CreatedAt:time.Unix(int64(i),0)},&types.Message{ID:request+"a",RequestID:request,Role:"assistant",Content:"received",IsCompleted:true,CreatedAt:time.Unix(int64(i),1)})
    };return out
}
func auditHistoryLog(t *testing.T,d map[string]any) {b,_:=json.Marshal(d);t.Log("AUDIT_JSON "+string(b))}

func TestAuditHistoryWindowSourceIdentity(t *testing.T) {
    rounds:=5; windowPairs:=conversationmemory.FetchMessageLimit(rounds)/2
    left,_:=buildGeneralAgentHistory(auditHistoryMessages(1,windowPairs),rounds)
    right,_:=buildGeneralAgentHistory(auditHistoryMessages(2,windowPairs+1),rounds)
    fact:=fmt.Sprintf("fact_%d",windowPairs)
    var a,b string;for _,m:=range left {if m.Content==fact {a=m.SourceID}};for _,m:=range right {if m.Content==fact {b=m.SourceID}}
    if a==""||b==""||a==b {t.Fatalf("expected ID to move with sliding window: %q %q",a,b)}
    auditHistoryLog(t,map[string]any{"experiment":"sliding_history_ids","configured_history_rounds":rounds,"fetched_messages":windowPairs*2,"unchanged_user_fact":fact,"before_source_id":a,"after_source_id":b})
}

func TestAuditInterruptedUserFactDisappears(t *testing.T) {
    messages:=auditHistoryMessages(1,3)
    messages[2].Content="authoritative-user-update"
    messages[3].IsCompleted=false
    history,ledger:=buildGeneralAgentHistory(messages,5)
    encoded,_:=json.Marshal(history)
    dropped:=!strings.Contains(string(encoded),"authoritative-user-update")&&!strings.Contains(ledger,"authoritative-user-update")
    messages[3].IsCompleted=true
    restoredHistory,restoredLedger:=buildGeneralAgentHistory(messages,5)
    restoredJSON,_:=json.Marshal(restoredHistory)
    restored:=strings.Contains(string(restoredJSON),"authoritative-user-update")&&strings.Contains(restoredLedger,"authoritative-user-update")
    if !dropped||!restored {t.Fatal("unexpected history inclusion behavior")}
    auditHistoryLog(t,map[string]any{"experiment":"interrupted_user_fact","user_text_unchanged":true,"assistant_incomplete_user_dropped":dropped,"only_completion_flag_changed_fact_restored":restored})
}

func TestAuditLongLedgerLosesLatestTailField(t *testing.T) {
    messages:=auditHistoryMessages(1,48)
    for i:=0;i<48;i++ {messages[2*i].Content=strings.Repeat("generic preserved user narrative. ",40)+fmt.Sprintf(" final_field_%d",i)}
    history,ledger:=buildGeneralAgentHistory(messages,5)
    full,_:=json.Marshal(history)
    lost:=!strings.Contains(ledger,"final_field_20")&&!strings.Contains(string(full),"final_field_20")
    if !lost {t.Fatal("expected truncation under fixed budget")}
    messages[40].Content="final_field_20 "+strings.Repeat("generic preserved user narrative. ",40)
    _,moved:=buildGeneralAgentHistory(messages,5)
    restored:=strings.Contains(moved,"final_field_20")
    if !restored {t.Fatal("expected beginning of the same old turn to survive")}
    auditHistoryLog(t,map[string]any{"experiment":"ledger_truncation","ledger_runes":len([]rune(ledger)),"input_turns":48,"older_turn_tail_field_missing_from_full_context":lost,"same_fact_at_start_of_same_turn_restored":restored,"limit_is_position_not_semantics":true})
}

func TestAuditArtifactsAreNotRehydratedFromAssistantHistory(t *testing.T) {
    messages:=auditHistoryMessages(1,1)
    messages[1].Content="File created: /api/v1/custom/general-agent/artifacts/audit-artifact/download"
    h,_:=buildGeneralAgentHistory(messages,5)
    original:=(&Service{}).originalInputFileSpecs(context.Background(),&types.QARequest{},"next-run")
    b,_:=json.Marshal(h)
    if !strings.Contains(string(b),"audit-artifact")||len(original)!=0 {t.Fatal("unexpected history file behavior")}
    auditHistoryLog(t,map[string]any{"experiment":"artifact_history","link_in_assistant_text":true,"next_turn_materialized_original_files":len(original),"assistant_content_only":true})
}
