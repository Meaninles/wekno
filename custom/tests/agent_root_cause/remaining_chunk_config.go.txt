package service

import (
    "encoding/json"
    "testing"

    "github.com/Tencent/WeKnora/internal/infrastructure/chunker"
    "github.com/Tencent/WeKnora/internal/types"
)

// Diagnostic assertions describe the observed implementation, not acceptance.
func TestRemainingParentStrategyObservation(t *testing.T) {
    for _, strategy := range []string{"auto", "heading", "heuristic", "recursive"} {
        cc := types.ChunkingConfig{Strategy: strategy, ChunkSize: 512, ChunkOverlap: 80, ParentChunkSize: 4096, ChildChunkSize: 384}
        base := buildSplitterConfigFromChunking(cc)
        parent, child := buildParentChildConfigs(cc, base)
        _, diag := chunker.SplitWithDiagnostics("# Generic section\n\nA paragraph.\n\n# Another section\n\nAnother paragraph.", child)
        data, _ := json.Marshal(map[string]interface{}{"configured": strategy, "base": base.Strategy, "parent": parent.Strategy, "child": child.Strategy, "selected_tier": diag.SelectedTier})
        t.Log(string(data))
        if base.Strategy != strategy || parent.Strategy != "" || child.Strategy != "" {
            t.Fatal("Observed configuration-loss behavior changed; re-evaluate this diagnosis")
        }
    }
}
