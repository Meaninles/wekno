func auditRetrievalLog(t *testing.T,d map[string]any){b,_:=json.Marshal(d);t.Log("AUDIT_JSON "+string(b))}
func TestAuditDedupIndependentOfArrival(t *testing.T) {
    tool:=&KnowledgeSearchTool{}
    low:=&searchResultWithMeta{SearchResult:&types.SearchResult{ID:"same-chunk",Content:"identical source evidence",Score:0.1}}
    high:=&searchResultWithMeta{SearchResult:&types.SearchResult{ID:"same-chunk",Content:"identical source evidence",Score:0.9}}
    a:=tool.deduplicateResults([]*searchResultWithMeta{low,high})
    b:=tool.deduplicateResults([]*searchResultWithMeta{high,low})
    if len(a)!=1||len(b)!=1||a[0].Score!=0.9||b[0].Score!=0.9 {t.Fatalf("unexpected dedup behavior")}
    auditRetrievalLog(t,map[string]any{"experiment":"dedup_arrival","same_candidates":true,"low_arrives_first_retained_score":a[0].Score,"high_arrives_first_retained_score":b[0].Score})
}

func TestAuditSiblingEvidencePreserved(t *testing.T) {
    tool:=&KnowledgeSearchTool{}
    a:=&searchResultWithMeta{SearchResult:&types.SearchResult{ID:"child-a",KnowledgeID:"doc",ParentChunkID:"parent",ChunkIndex:1,Content:"unrelated introductory information",Score:0.1}}
    b:=&searchResultWithMeta{SearchResult:&types.SearchResult{ID:"child-b",KnowledgeID:"doc",ParentChunkID:"parent",ChunkIndex:2,Content:"the distinct requested evidence is in this sibling",Score:0.9}}
    forward:=tool.deduplicateResults([]*searchResultWithMeta{a,b})
    reverse:=tool.deduplicateResults([]*searchResultWithMeta{b,a})
    if len(forward)!=2||len(reverse)!=2||forward[0].ID!=reverse[0].ID {t.Fatal("unexpected sibling behavior")}
    auditRetrievalLog(t,map[string]any{"experiment":"sibling_evidence_loss","same_parent_different_content":true,"forward_keeps":forward[0].ID,"reverse_keeps":reverse[0].ID,"dropped_before_reranker":false})
}

func TestAuditBelowThresholdRecallFloor(t *testing.T) {
    scores:=[]rerank.RankResult{{Index:0,RelevanceScore:0.09},{Index:1,RelevanceScore:0.08},{Index:2,RelevanceScore:0.07},{Index:3,RelevanceScore:0.06}}
    kept:=filterKnowledgeSearchScores(scores,4,0.5)
    if len(kept)!=0 {t.Fatalf("below-threshold retained: %d",len(kept))}
    auditRetrievalLog(t,map[string]any{"experiment":"recall_floor","threshold":0.5,"scores":scores,"retained_count":len(kept),"retained_all_below_threshold":true})
}

func TestAuditSchemaValidationRejectsUnexecutableArguments(t *testing.T) {
    tool:=NewGetDocumentInfoTool(nil,nil,nil)
    for _,raw:=range []string{`{}`,`{"knowledge_ids":[]}`,`{"knowledge_ids":[123]}`,`{"knowledge_ids":`} {
        schemaErrors:=ValidateParams(json.RawMessage(raw),tool.Parameters())
        result,err:=tool.Execute(context.Background(),json.RawMessage(raw))
        if len(schemaErrors)==0||err==nil||result.Success {t.Fatalf("expected schema accepted, execution rejected: %s",raw)}
        auditRetrievalLog(t,map[string]any{"experiment":"tool_contract","input":raw,"validator_accepts":false,"executor_error":result.Error})
    }
}

type auditUnavailableKB struct {interfaces.KnowledgeBaseService}
func (a auditUnavailableKB) GetKnowledgeBasesByIDsOnly(context.Context,[]string)([]*types.KnowledgeBase,error) {return nil,errors.New("controlled repository unavailable")}
func (a auditUnavailableKB) ResolveEmbeddingModelKeys(context.Context,[]*types.KnowledgeBase) map[string]string {return nil}
func (a auditUnavailableKB) HybridSearch(context.Context,string,types.SearchParams)([]*types.SearchResult,error) {return nil,errors.New("controlled retrieval connection failure")}

func TestAuditRetrievalFailureBecomesSuccessfulNoEvidence(t *testing.T) {
    tool:=&KnowledgeSearchTool{knowledgeBaseService:auditUnavailableKB{}}
    results,searchErr:=tool.concurrentSearchByTargets(context.Background(),[]string{"arbitrary evidence request"},types.SearchTargets{&types.SearchTarget{Type:types.SearchTargetTypeKnowledgeBase,KnowledgeBaseID:"audit-kb",TenantID:10000}},5,0,0,nil)
    if searchErr==nil || len(results)!=0 {t.Fatalf("failure was collapsed into empty success: %v",searchErr)}
}
